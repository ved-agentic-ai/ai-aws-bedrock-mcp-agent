Live AWS Agentic Architecture

End-to-end event-driven architecture decoupling LLM reasoning (Bedrock) from tool computation via Model Context Protocol (MCP).

🧠 1. Amazon Bedrock (Nova Micro / Claude)
High-speed Foundation Model reasoning engine. Bedrock evaluates incoming user prompts, determines intent, and generates a structured toolUse stop-reason payload whenever external actions are required.
🔌 2. Model Context Protocol (MCP) Standard
Open JSON-RPC specification (tools/list & tools/call) decoupling intelligence from tool execution. Enables swapping underlying AI models without altering backend business logic.
⚡ 3. AWS Lambda Serverless Functions
Dual Lambda architecture:
• Agent Runtime Lambda: Manages prompt processing, Bedrock API communication, and agent loop execution.
• MCP Tool Server Lambda: Executes database queries and tool invocations safely in an isolated sandbox.
💾 4. Amazon DynamoDB On-Demand
Pay-per-request session memory database storing short-term and long-term conversation history. Configured with automatic 24-hour TTL expiration for zero idle cost.
🚀 5. AWS CloudFormation (IaC)
100% Infrastructure-as-Code automation template provisioning IAM roles, Lambda functions, DynamoDB tables, and API Gateway endpoints with 1-click deploy and clean teardown.

import json import os import boto3 bedrock_runtime = boto3.client('bedrock-runtime') lambda_client = boto3.client('lambda') def call_mcp_tool_server(action, name=None, arguments=None): payload = {"action": action} if name: payload["name"] = name if arguments: payload["arguments"] = arguments res = lambda_client.invoke( FunctionName=os.environ['MCP_SERVER_FUNCTION'], InvocationType='RequestResponse', Payload=json.dumps({"body": json.dumps(payload)}) ) res_payload = json.loads(res['Payload'].read().decode('utf-8')) return json.loads(res_payload.get('body', '{}')) def handler(event, context): body = json.loads(event.get('body', '{}')) if isinstance(event.get('body'), str) else event.get('body', {}) user_prompt = body.get('prompt', 'Hello AWS Bedrock!') session_id = body.get('session_id', 'session_101') # Step 1: Discover available MCP tools mcp_tools_res = call_mcp_tool_server('tools/list') tools_list = mcp_tools_res.get('tools', []) bedrock_tools = [] for t in tools_list: bedrock_tools.append({ "toolSpec": { "name": t["name"], "description": t["description"], "inputSchema": { "json": t["input_schema"] } } }) system_prompt = [{"text": "You are Nova AI Assistant. Address the user directly in first-person with clean Markdown formatting."}] model_id = os.environ.get('BEDROCK_MODEL_ID', 'us.amazon.nova-micro-v1:0') messages = [{"role": "user", "content": [{"text": user_prompt}]}] # Step 2: First Converse API call response = bedrock_runtime.converse( modelId=model_id, messages=messages, system=system_prompt, toolConfig={"tools": bedrock_tools} if bedrock_tools else None ) output_message = response['output']['message'] stop_reason = response.get('stopReason') if stop_reason == 'tool_use': tool_requests = [c for c in output_message['content'] if 'toolUse' in c] tool_results = [] for tr in tool_requests: t_use = tr['toolUse'] t_args = t_use['input'] if 'session_id' not in t_args: t_args['session_id'] = session_id exec_res = call_mcp_tool_server('tools/call', name=t_use['name'], arguments=t_args) tool_results.append({ "toolResult": { "toolUseId": t_use['toolUseId'], "content": [{"json": exec_res.get('result', {})}] } }) messages.append(output_message) messages.append({"role": "user", "content": tool_results}) # Step 3: Second Converse API call to synthesize final response final_response = bedrock_runtime.converse( modelId=model_id, messages=messages, system=system_prompt, toolConfig={"tools": bedrock_tools} if bedrock_tools else None ) final_content = final_response['output']['message']['content'] final_text = "".join([b.get('text', '') for b in final_content if 'text' in b]) return { "statusCode": 200, "body": json.dumps({"response": final_text, "mcp_tools_executed": [tr['toolUse']['name'] for tr in tool_requests]}) } else: return {"statusCode": 200, "body": json.dumps({"response": output_message['content'][0]['text']})}
import json import os import boto3 import time from boto3.dynamodb.conditions import Key dynamodb = boto3.resource('dynamodb') def list_mcp_tools(): return [ { "name": "query_database", "description": "Query session history from DynamoDB", "input_schema": { "type": "object", "properties": { "session_id": {"type": "string"} }, "required": ["session_id"] } }, { "name": "save_session_data", "description": "Store state into session memory", "input_schema": { "type": "object", "properties": { "session_id": {"type": "string"}, "content": {"type": "string"} }, "required": ["session_id", "content"] } } ] def execute_mcp_tool(tool_name, arguments): table = dynamodb.Table(os.environ['TABLE_NAME']) session_id = str(arguments.get("session_id", "session_101")) if tool_name == "query_database": res = table.query( KeyConditionExpression=Key('SessionId').eq(session_id), ScanIndexForward=False, Limit=10 ) items = res.get('Items', []) for item in items: for k, v in list(item.items()): if hasattr(v, '__float__'): item[k] = float(v) return {"status": "success", "items": items, "count": len(items)} elif tool_name == "save_session_data": content = str(arguments.get("content", "User session memory entry")) now = int(time.time()) table.put_item(Item={ 'SessionId': session_id, 'Timestamp': now, 'Content': content, 'TTL': now + 86400 }) return {"status": "success", "message": f"Successfully stored session item for {session_id} in DynamoDB table!"} return {"error": f"Tool {tool_name} not found"} def handler(event, context): body = json.loads(event.get('body', '{}')) if isinstance(event.get('body'), str) else event.get('body', {}) action = body.get('action', 'tools/list') if action == 'tools/list': return {"statusCode": 200, "body": json.dumps({"tools": list_mcp_tools()})} elif action == 'tools/call': result = execute_mcp_tool(body.get('name'), body.get('arguments', {})) return {"statusCode": 200, "body": json.dumps({"result": result})}
